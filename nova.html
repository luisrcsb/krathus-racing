function gerarFiltrosDinamicos() { const batContainer = document.getElementById('filtro-baterias-container'); const pilContainer = document.getElementById('filtro-pilotos-container'); if (!batContainer || !pilContainer) return; const pilotosSet = new Set(); listaJsonsCache.forEach(arq => { if (arq.dados && Array.isArray(arq.dados)) { arq.dados.forEach(d => { if (d.piloto && d.laps && d.laps.length > 0) { let pReal = d.piloto; let safeKey = pReal.replace(/[.#$\/\[\]]/g, "_"); // Se este nome foi mesclado em outro, aponta para o destino principal if (mesclagensCache[safeKey]) { pReal = mesclagensCache[safeKey]; } pilotosSet.add(pReal); } }); } }); Object.keys(pilotosMetadadosCache).forEach(p => { let safeKey = p.replace(/[.#$\/\[\]]/g, "_"); let pReal = mesclagensCache[safeKey] || p; pilotosSet.add(pReal); }); if (listaJsonsCache.length === 0) { batContainer.innerHTML = `Nenhum arquivo.`; pilContainer.innerHTML = `Nenhum piloto.`; return; } let bateriasSelecionadasAntigas = Array.from(document.querySelectorAll('.filter-bateria:checked')).map(cb => cb.value); let pilotosSelecionadosAntigos = Array.from(document.querySelectorAll('.filter-piloto:checked')).map(cb => cb.value); let primeiraVezBat = bateriasSelecionadasAntigas.length === 0; let primeiraVezPil = pilotosSelecionadosAntigos.length === 0; batContainer.innerHTML = listaJsonsCache.map(arq => { let key = arq.firebaseKey; let sessaoOriginal = arq.sessao || (arq.dados && arq.dados[0] ? arq.dados[0].sessao : key); let sessaoFormatada = formatarNomeSessao(sessaoOriginal); let marcado = (primeiraVezBat || bateriasSelecionadasAntigas.includes(key)) ? 'checked' : ''; return ``; }).join(''); // Filtro rigoroso: remove qualquer nome que seja chave de origem em mesclagensCache // e garante que nomes curtos isolados (como "DJ" ou "Raphael") não apareçam se já existe uma versão completa mapeada. let listaNomesUnicos = Array.from(pilotosSet).filter(p => { let sKey = p.replace(/[.#$\/\[\]]/g, "_"); // Se este exato nome é uma chave de origem que foi mesclada para outro, ignora ele if (mesclagensCache[sKey]) return false; // Se o nome for curto (ex: "DJ" ou "Raphael"), verifica se há um nome completo correspondente na lista let isNomeCurtoIsolado = Array.from(pilotosSet).some(outroP => outroP !== p && outroP.includes(p) && outroP.length > p.length); if (isNomeCurtoIsolado && (p.length <= 8 && !pilotosMetadadosCache[p]?.apelido)) { return false; } return true; }); pilContainer.innerHTML = listaNomesUnicos.sort().map(piloto => { let meta = pilotosMetadadosCache[piloto] || {}; let labelExib = meta.apelido ? `${piloto} (${meta.apelido})` : piloto; let marcado = primeiraVezPil || pilotosSelecionadosAntigos.includes(piloto) ? 'checked' : ''; return ``; }).join(''); }
