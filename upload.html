// Função robusta para processar o PDF do ZRound e capturar TODAS as voltas sem limite window.processarUploadEEnviarNuvem = async function() { const input = document.getElementById('input-pdf-upload'); if (!input.files || input.files.length === 0) { alert("Por favor, selecione ao menos um arquivo PDF do ZRound."); return; } if (!db) { alert("Erro: Firebase não conectado."); return; } for (let i = 0; i < input.files.length; i++) { const arquivo = input.files[i]; const nomeArq = arquivo.name; const sessaoNome = nomeArq.replace(/\.[^/.]+$/, "").trim(); const batKey = "bat_" + sessaoNome.toLowerCase().replace(/[^a-z0-9]/g, "_"); try { const textoPdf = await lerTextoDoPdf(arquivo); const dadosExtraidos = parsearRelatorioZRound(textoPdf, sessaoNome, batKey); const novoRegistro = { bateriaKey: batKey, sessao: sessaoNome, id: Date.now() + i, dados: dadosExtraidos }; await db.ref('baterias').push(novoRegistro); console.log(`Sessão "${sessaoNome}" salva com todas as voltas na nuvem!`); } catch (err) { console.error("Erro ao processar o arquivo " + nomeArq, err); alert("Erro ao processar o arquivo " + nomeArq + ": " + err.message); } } alert("Relatórios processados com sucesso e todas as voltas foram enviadas para a nuvem!"); input.value = ""; if (typeof atualizarListaUploadHistorico === 'function') atualizarListaUploadHistorico(); }; function lerTextoDoPdf(arquivo) { return new Promise((resolve, reject) => { const reader = new FileReader(); reader.onload = function() { resolve(reader.result); }; reader.onerror = reject; reader.readAsText(arquivo, "UTF-8"); }); } // Parser blindado para varrer e capturar 100% das voltas do PDF do ZRound function parsearRelatorioZRound(texto, sessaoNome, batKey) { let pilotosMap = {}; let linhas = texto.split(/\r?\n/); // 1. Identifica os nomes dos pilotos no cabeçalho da tabela de resumo linhas.forEach(linha => { let partes = linha.split('|').map(p => p.trim()); if (partes.length >= 5) { let possivelPiloto = partes[1]; let possivelPos = partes[2]; let possivelVoltas = partes[3]; let possivelTempoTot = partes[4]; if (possivelPos && (possivelPos.endsWith('º') || !isNaN(possivelPos)) && isNaN(possivelPiloto)) { let pilotoNome = possivelPiloto; let posStr = possivelPos.includes('º') ? possivelPos : possivelPos + 'º'; pilotosMap[pilotoNome] = { pos: posStr, piloto: pilotoNome, bateriaKey: batKey, sessao: sessaoNome, voltasTotais: parseInt(possivelVoltas) || 0, tempoTotal: possivelTempoTot || "00:00,000", melhorVoltaVal: 0, melhorVoltaTxt: "00,000s", mediaVal: 0, mediaTxt: "00,000s", desvioVal: 0, desvio: "±0,000s", badgeColor: "#ffb703", laps: [] }; } } }); let pilotoKeys = Object.keys(pilotosMap); // 2. Varre linha por linha procurando a tabela volta a volta de forma irrestrita linhas.forEach(linha => { let partes = linha.split('|').map(p => p.trim()); // Linhas de voltas começam com o número da volta (ex: "1", "2", ... "500") if (partes.length >= 2 && !isNaN(partes[0]) && partes[0] !== "") { let numVolta = parseInt(partes[0], 10); // Para cada piloto mapeado, verifica se há tempo correspondente na coluna da tabela pilotoKeys.forEach((pName, idx) => { let colunaIndex = idx + 1; if (partes[colunaIndex] && partes[colunaIndex] !== "") { let celula = partes[colunaIndex]; // Remove eventuais marcadores adicionais e limpa para converter o formato de tempo let lapTexto = celula.split('-')[0].trim(); let tempoNum = parseFloat(lapTexto.replace(',', '.')); if (!isNaN(tempoNum) && tempoNum > 0) { // Insere o tempo garantindo que a posição do array corresponda exatamente à volta pilotosMap[pName].laps[numVolta - 1] = tempoNum; } } }); } }); // 3. Limpa eventuais buracos de array e calcula métricas finais para cada piloto let resultadoFinal = []; let coresBadge = ["#ffb703", "#e76f51", "#2ec4b6", "#3a86ff", "#e63946"]; let contador = 0; pilotoKeys.forEach(pName => { let pObj = pilotosMap[pName]; // Remove valores vazios caso haja saltos pObj.laps = pObj.laps.filter(val => val !== undefined && !isNaN(val)); if (pObj.laps.length > 0) { let melhor = Math.min(...pObj.laps); let soma = pObj.laps.reduce((a, b) => a + b, 0); let media = soma / pObj.laps.length; let variancia = pObj.laps.reduce((sum, t) => sum + Math.pow(t - media, 2), 0) / pObj.laps.length; let desvio = Math.sqrt(variancia); pObj.melhorVoltaVal = parseFloat(melhor.toFixed(3)); pObj.melhorVoltaTxt = melhor.toFixed(3).replace('.', ',') + 's'; pObj.mediaVal = parseFloat(media.toFixed(3)); pObj.mediaTxt = media.toFixed(3).replace('.', ',') + 's'; pObj.desvioVal = parseFloat(desvio.toFixed(3)); pObj.desvio = `±${desvio.toFixed(3).replace('.', ',')}s`; pObj.voltasTotais = pObj.laps.length; } pObj.badgeColor = coresBadge[contador % coresBadge.length]; contador++; resultadoFinal.push(pObj); }); return resultadoFinal; }
