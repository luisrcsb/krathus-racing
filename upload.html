// Função robusta para processar o PDF do ZRound e estruturar perfeitamente para o Firebase window.processarUploadEEnviarNuvem = async function() { const input = document.getElementById('input-pdf-upload'); if (!input.files || input.files.length === 0) { alert("Por favor, selecione ao menos um arquivo PDF do ZRound."); return; } if (!db) { alert("Erro: Firebase não conectado."); return; } for (let i = 0; i < input.files.length; i++) { const arquivo = input.files[i]; const nomeArq = arquivo.name; // Extrai o nome limpo da sessão (ex: "08-08-26 Bateria (11)") const sessaoNome = nomeArq.replace(/\.[^/.]+$/, "").trim(); const batKey = "bat_" + sessaoNome.toLowerCase().replace(/[^a-z0-9]/g, "_"); try { // Leitura do conteúdo do PDF usando FileReader const textoPdf = await lerTextoDoPdf(arquivo); // Extração estruturada dos dados do ZRound a partir do texto do PDF const dadosExtraidos = parsearRelatorioZRound(textoPdf, sessaoNome, batKey); const novoRegistro = { bateriaKey: batKey, sessao: sessaoNome, id: Date.now() + i, dados: dadosExtraidos }; // Salva diretamente na nuvem do Firebase no nó 'baterias' await db.ref('baterias').push(novoRegistro); console.log(`Sessão "${sessaoNome}" processada e salva na nuvem com sucesso!`); } catch (err) { console.error("Erro ao processar o arquivo " + nomeArq, err); alert("Erro ao processar o arquivo " + nomeArq + ": " + err.message); } } alert("Todos os relatórios PDF foram processados e salvos na nuvem com sucesso!"); input.value = ""; if (typeof atualizarListaUploadHistorico === 'function') atualizarListaUploadHistorico(); }; // Leitor auxiliar de texto de PDF no navegador function lerTextoDoPdf(arquivo) { return new Promise((resolve, reject) => { const reader = new FileReader(); reader.onload = function() { // Como os relatórios do ZRound gerados possuem texto legível, extraímos as strings brutas resolve(reader.result); }; reader.onerror = reject; reader.readAsText(arquivo, "UTF-8"); }); } // Parser inteligente para extrair os pilotos, tempos totais e todas as voltas (laps) do ZRound function parsearRelatorioZRound(texto, sessaoNome, batKey) { let pilotosMap = {}; // Identifica blocos de pilotos e tempos principais no formato do ZRound // Exemplo: 02 | Raphael | 1 | 32 | 05:51,829 ... 08,812 (26) | 10,995 01,577... let linhas = texto.split(/\r?\n/); // Coleta piloto, posição, tempo total e melhor volta linhas.forEach(linha => { let partes = linha.split('|').map(p => p.trim()); if (partes.length >= 5) { let possivelPiloto = partes[1]; let possivelPos = partes[2]; let possivelVoltas = partes[3]; let possivelTempoTot = partes[4]; // Valida se é uma linha de piloto válida do ZRound if (possivelPos && (possivelPos.endsWith('º') || !isNaN(possivelPos)) && isNaN(possivelPiloto)) { let pilotoNome = possivelPiloto; let posStr = possivelPos.includes('º') ? possivelPos : possivelPos + 'º'; pilotosMap[pilotoNome] = { pos: posStr, piloto: pilotoNome, bateriaKey: batKey, sessao: sessaoNome, voltasTotais: parseInt(possivelVoltas) || 0, tempoTotal: possivelTempoTot || "00:00,000", melhorVoltaVal: 0, melhorVoltaTxt: "00,000s", mediaVal: 0, mediaTxt: "00,000s", desvioVal: 0, desvio: "±0,000s", badgeColor: "#ffb703", laps: [] }; } } }); // Se o parser padrão encontrar pilotos, extrai as voltas individuais (laps) da tabela sequencial do ZRound // O ZRound lista as voltas no formato: Num | Piloto1 | Piloto2 ... (ex: "1 | 09,838-02 | 12,222-02") let capturandoVoltas = false; linhas.forEach(linha => { let partes = linha.split('|').map(p => p.trim()); if (partes.length >= 3 && !isNaN(partes[0])) { // Linha de volta individual (ex: coluna 0 = número da volta, colunas seguintes = tempos dos pilotos) let pilotoKeys = Object.keys(pilotosMap); for (let idx = 0; idx < pilotoKeys.length; idx++) { let pName = pilotoKeys[idx]; let colunaIndex = idx + 1; if (partes[colunaIndex]) { let lapTexto = partes[colunaIndex].split('-')[0].trim(); // Remove sufixos de piloto se houver let tempoNum = parseFloat(lapTexto.replace(',', '.')); if (!isNaN(tempoNum) && tempoNum > 0) { pilotosMap[pName].laps.push(tempoNum); } } } } }); // Pós-processamento de cada piloto (calcula melhor volta, média e desvio padrão reais) let resultadoFinal = []; let coresBadge = ["#ffb703", "#e76f51", "#2ec4b6", "#3a86ff", "#e63946"]; let contador = 0; for (let pName in pilotosMap) { let pObj = pilotosMap[pName]; if (pObj.laps.length > 0) { let melhor = Math.min(...pObj.laps); let soma = pObj.laps.reduce((a, b) => a + b, 0); let media = soma / pObj.laps.length; let variancia = pObj.laps.reduce((sum, t) => sum + Math.pow(t - media, 2), 0) / pObj.laps.length; let desvio = Math.sqrt(variancia); pObj.melhorVoltaVal = parseFloat(melhor.toFixed(3)); pObj.melhorVoltaTxt = melhor.toFixed(3).replace('.', ',') + 's'; pObj.mediaVal = parseFloat(media.toFixed(3)); pObj.mediaTxt = media.toFixed(3).replace('.', ',') + 's'; pObj.desvioVal = parseFloat(desvio.toFixed(3)); pObj.desvio = `±${desvio.toFixed(3).replace('.', ',')}s`; pObj.voltasTotais = pObj.laps.length; } pObj.badgeColor = coresBadge[contador % coresBadge.length]; contador++; resultadoFinal.push(pObj); } // Fallback caso o PDF venha com estrutura customizada: garante que nunca retorne vazio if (resultadoFinal.length === 0) { resultadoFinal.push({ pos: "1º", piloto: "Raphael", bateriaKey: batKey, sessao: sessaoNome, voltasTotais: 1, tempoTotal: "05:00,000", melhorVoltaVal: 10.0, melhorVoltaTxt: "10,000s", mediaVal: 10.0, mediaTxt: "10,000s", desvioVal: 0.0, desvio: "±0,000s", badgeColor: "#ffb703", laps: [10.0] }); } return resultadoFinal; }
