function parsearTextoConvertido(texto, sessaoNome, batKey) { let pilotosMap = {}; let linhas = texto.split(/\r?\n/).map(l => l.trim()).filter(Boolean); let modoResumo = false; let modoVoltas = false; let ordemPilotos = []; // 1. Varredura inteligente baseada nas seções reais do relatório do ZRound linhas.forEach(linha => { let partes = linha.split(/[\s|]+/).filter(Boolean); if (partes.length === 0) return; // Detecta o início da tabela de resumo dos pilotos if (partes.includes("Piloto") && partes.includes("Voltas")) { modoResumo = true; modoVoltas = false; return; } // Detecta o início da tabela de voltas individuais if (linha.startsWith("Voltas por piloto") || (partes.includes("Num.") && modoResumo)) { modoResumo = false; modoVoltas = true; return; } // Se estivermos na tabela de resumo, captura o nome exato do piloto e dados gerais if (modoResumo && /^\d{2}$/.test(partes[0])) { let numOrdem = partes[0]; let possivelPiloto = partes[1]; let possivelPos = partes[2]; let possivelVoltas = partes[3]; let possivelTempo = partes[4] || "00:00,000"; if (isNaN(possivelPiloto) && /^\d+º?$/.test(possivelPos)) { let pilotoNome = possivelPiloto; let posStr = possivelPos.includes('º') ? possivelPos : possivelPos + 'º'; pilotosMap[pilotoNome] = { pos: posStr, piloto: pilotoNome, bateriaKey: batKey, sessao: sessaoNome, voltasTotais: parseInt(possivelVoltas) || 0, tempoTotal: possivelTempo, melhorVoltaVal: 0, melhorVoltaTxt: "00,000s", mediaVal: 0, mediaTxt: "00,000s", desvioVal: 0, desvio: "±0,000s", badgeColor: "#ffb703", laps: [] }; ordemPilotos.push(pilotoNome); } } // Se estivermos na tabela de voltas, lê o número da volta e distribui para os pilotos na ordem correta if (modoVoltas && /^\d+$/.test(partes[0])) { let numVolta = parseInt(partes[0], 10); let colIdx = 1; ordemPilotos.forEach(pName => { while (colIdx < partes.length) { let token = partes[colIdx]; // Identifica tempos válidos (ex: 14,629 ou 03:25,1) ignorando hífens ou traços if (token && token !== "-" && (/\d+[,.]\d+/.test(token) || /\d+:\d+[,.]\d+/.test(token))) { let tempoLimpo = token.split('-')[0].trim(); let tempoNum = 0; if (tempoLimpo.includes(':')) { let partesTempo = tempoLimpo.split(':'); let minutos = parseInt(partesTempo[0], 10); let segundos = parseFloat(partesTempo[1].replace(',', '.')); tempoNum = (minutos * 60) + segundos; } else { tempoNum = parseFloat(tempoLimpo.replace(',', '.')); } if (!isNaN(tempoNum) && tempoNum > 0) { pilotosMap[pName].laps[numVolta - 1] = parseFloat(tempoNum.toFixed(3)); colIdx++; break; } } colIdx++; } }); } }); let resultadoFinal = []; let coresBadge = ["#ffb703", "#e76f51", "#2ec4b6", "#3a86ff", "#e63946"]; let contador = 0; Object.keys(pilotosMap).forEach(pName => { let pObj = pilotosMap[pName]; pObj.laps = pObj.laps.filter(val => val !== undefined && !isNaN(val)); if (pObj.laps.length > 0) { let melhor = Math.min(...pObj.laps); let soma = pObj.laps.reduce((a, b) => a + b, 0); let media = soma / pObj.laps.length; let variancia = pObj.laps.reduce((sum, t) => sum + Math.pow(t - media, 2), 0) / pObj.laps.length; let desvio = Math.sqrt(variancia); pObj.melhorVoltaVal = parseFloat(melhor.toFixed(3)); pObj.melhorVoltaTxt = melhor.toFixed(3).replace('.', ',') + 's'; pObj.mediaVal = parseFloat(media.toFixed(3)); pObj.mediaTxt = media.toFixed(3).replace('.', ',') + 's'; pObj.desvioVal = parseFloat(desvio.toFixed(3)); pObj.desvio = `±${desvio.toFixed(3).replace('.', ',')}s`; pObj.voltasTotais = pObj.laps.length; } pObj.badgeColor = coresBadge[contador % coresBadge.length]; contador++; resultadoFinal.push(pObj); }); return resultadoFinal; }
