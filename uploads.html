function parsearTextoConvertido(texto, sessaoNome, batKey) { let pilotosMap = {}; let linhas = texto.split(/\r?\n/).map(l => l.trim()).filter(Boolean); // 1. Extração cirúrgica dos pilotos na tabela de resumo inicial do ZRound // Identifica linhas que possuem o formato: [Numero/Ordem] | [Nome do Piloto] | [Posicao] | [Voltas] | [TempoTotal] linhas.forEach(linha => { let partes = linha.split(/[\s|]+/).filter(Boolean); // Exemplo: ["02", "DJ", "1", "29", "06:11,204"] ou ["01", "Ronaldo", "2", "26", ...] if (partes.length >= 5) { let idOrdem = partes[0]; let possivelPiloto = partes[1]; let possivelPos = partes[2]; let possivelVoltas = partes[3]; let possivelTempo = partes[4]; // Garante que o ID de ordem é um número de linha e o piloto NÃO é um número if (/^\d{2}$/.test(idOrdem) && isNaN(possivelPiloto) && /^\d+º?$/.test(possivelPos)) { let pilotoNome = possivelPiloto; let posStr = possivelPos.includes('º') ? possivelPos : possivelPos + 'º'; pilotosMap[pilotoNome] = { pos: posStr, piloto: pilotoNome, bateriaKey: batKey, sessao: sessaoNome, voltasTotais: parseInt(possivelVoltas) || 0, tempoTotal: possivelTempo, melhorVoltaVal: 0, melhorVoltaTxt: "00,000s", mediaVal: 0, mediaTxt: "00,000s", desvioVal: 0, desvio: "±0,000s", badgeColor: "#ffb703", laps: [] }; } } }); let pilotoNomes = Object.keys(pilotosMap); // 2. Extração rigorosa das voltas individuais na tabela inferior (iniciada por número de volta) let capturandoVoltas = false; linhas.forEach(linha => { let partes = linha.split(/[\s|]+/).filter(Boolean); if (partes.length > 0 && partes[0] === "Num.") { capturandoVoltas = true; return; } if (capturandoVoltas && partes.length >= 2 && /^\d+$/.test(partes[0])) { let numVolta = parseInt(partes[0], 10); let colIdx = 1; pilotoNomes.forEach(pName => { while (colIdx < partes.length) { let token = partes[colIdx]; // Aceita formatos de tempo válidos em segundos (ex: 14,629 ou 03:25,1) if (token && (/\d+[,.]\d+/.test(token) || /\d+:\d+[,.]\d+/.test(token))) { let tempoLimpo = token.split('-')[0].trim(); let tempoNum = 0; if (tempoLimpo.includes(':')) { let partesTempo = tempoLimpo.split(':'); let minutos = parseInt(partesTempo[0], 10); let segundos = parseFloat(partesTempo[1].replace(',', '.')); tempoNum = (minutos * 60) + segundos; } else { tempoNum = parseFloat(tempoLimpo.replace(',', '.')); } if (!isNaN(tempoNum) && tempoNum > 0) { pilotosMap[pName].laps[numVolta - 1] = parseFloat(tempoNum.toFixed(3)); colIdx++; break; } } colIdx++; } }); } }); // 3. Processamento de estatísticas finais (melhor volta, média, desvio padrão) let resultadoFinal = []; let coresBadge = ["#ffb703", "#e76f51", "#2ec4b6", "#3a86ff", "#e63946"]; let contador = 0; pilotoNomes.forEach(pName => { let pObj = pilotosMap[pName]; pObj.laps = pObj.laps.filter(val => val !== undefined && !isNaN(val)); if (pObj.laps.length > 0) { let melhor = Math.min(...pObj.laps); let soma = pObj.laps.reduce((a, b) => a + b, 0); let media = soma / pObj.laps.length; let variancia = pObj.laps.reduce((sum, t) => sum + Math.pow(t - media, 2), 0) / pObj.laps.length; let desvio = Math.sqrt(variancia); pObj.melhorVoltaVal = parseFloat(melhor.toFixed(3)); pObj.melhorVoltaTxt = melhor.toFixed(3).replace('.', ',') + 's'; pObj.mediaVal = parseFloat(media.toFixed(3)); pObj.mediaTxt = media.toFixed(3).replace('.', ',') + 's'; pObj.desvioVal = parseFloat(desvio.toFixed(3)); pObj.desvio = `±${desvio.toFixed(3).replace('.', ',')}s`; pObj.voltasTotais = pObj.laps.length; } pObj.badgeColor = coresBadge[contador % coresBadge.length]; contador++; resultadoFinal.push(pObj); }); return resultadoFinal; }
