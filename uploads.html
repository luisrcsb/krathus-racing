window.processarUploadEEnviarNuvem = async function() { const input = document.getElementById('input-pdf-upload'); if (!input.files || input.files.length === 0) { alert("Por favor, selecione ao menos um arquivo PDF."); return; } if (!db) { alert("Erro: Firebase não conectado."); return; } let arquivosNovosEnviados = 0; let arquivosDuplicados = []; for (let i = 0; i < input.files.length; i++) { const arquivo = input.files[i]; const nomeArq = arquivo.name; const sessaoNome = nomeArq.replace(/\.[^/.]+$/, "").trim(); const batKey = "bat_" + sessaoNome.toLowerCase().replace(/[^a-z0-9]/g, "_"); // 🛡️ Filtro Anticuplicação: Verifica se já existe uma sessão com o mesmo nome na nuvem const jaExiste = bateriasNuvemCache.some(item => (item.sessao && item.sessao.toLowerCase() === sessaoNome.toLowerCase()) || item.bateriaKey === batKey ); if (jaExiste) { arquivosDuplicados.push(nomeArq); continue; // Pula este arquivo e não envia } try { const textoPdf = await converterPdfParaTexto(arquivo); const dadosExtraidos = parsearTextoConvertido(textoPdf, sessaoNome, batKey); const novoRegistro = { bateriaKey: batKey, sessao: sessaoNome, id: Date.now() + i, dados: dadosExtraidos }; await db.ref('baterias').push(novoRegistro); arquivosNovosEnviados++; } catch (err) { console.error("Erro ao processar PDF:", err); alert("Erro ao converter o arquivo " + nomeArq + ": " + err.message); } } // Feedback claro para você if (arquivosNovosEnviados > 0 && arquivosDuplicados.length === 0) { alert("Relatório(s) PDF processado(s) e salvo(s) na nuvem com sucesso!"); } else if (arquivosNovosEnviados > 0 && arquivosDuplicados.length > 0) { alert(`Processado com sucesso! Porém, ${arquivosDuplicados.length} arquivo(s) foram ignorados por já existirem na nuvem: ${arquivosDuplicados.join(', ')}`); } else if (arquivosDuplicados.length > 0) { alert(`⚠️ Envio cancelado: O(s) arquivo(s) "${arquivosDuplicados.join(', ')}" já está(ão) cadastrado(s) na nuvem! Não foram feitas duplicações.`); } input.value = ""; };
