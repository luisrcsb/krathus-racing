function parsearRelatorioZRound(texto, sessaoNome, batKey) { let pilotosMap = {}; let linhas = texto.split(/\r?\n/); // 1. Identifica os pilotos no cabeçalho da tabela de classificação linhas.forEach(linha => { let partes = linha.split('|').map(p => p.trim()); if (partes.length >= 5) { let numOrdem = partes[0]; let possivelPiloto = partes[1]; let possivelPos = partes[2]; let possivelVoltas = partes[3]; let possivelTempoTot = partes[4]; // Detecta linhas de pilotos válidas (ex: "02 | Raphael | 1 | 32 | 05:51,829") if ((numOrdem === "02" || numOrdem === "03" || /^\d+$/.test(numOrdem)) && isNaN(possivelPiloto) && possivelPos) { let pilotoNome = possivelPiloto; let posStr = possivelPos.includes('º') ? possivelPos : possivelPos + 'º'; pilotosMap[pilotoNome] = { pos: posStr, piloto: pilotoNome, bateriaKey: batKey, sessao: sessaoNome, voltasTotais: parseInt(possivelVoltas) || 0, tempoTotal: possivelTempoTot || "00:00,000", melhorVoltaVal: 0, melhorVoltaTxt: "00,000s", mediaVal: 0, mediaTxt: "00,000s", desvioVal: 0, desvio: "±0,000s", badgeColor: "#ffb703", laps: [] }; } } }); let pilotoNomes = Object.keys(pilotosMap); // 2. Extrai rigorosamente todas as voltas da tabela volta a volta (ex: "1 | 09,838-02 | 12,222-02") linhas.forEach(linha => { let partes = linha.split('|').map(p => p.trim()); // Se a linha começa com o número da volta if (partes.length >= 2 && /^\d+$/.test(partes[0])) { let numVolta = parseInt(partes[0], 10); pilotoNomes.forEach((pName, idx) => { let colunaIndex = idx + 1; if (partes[colunaIndex] && partes[colunaIndex] !== "" && partes[colunaIndex] !== "-") { let celula = partes[colunaIndex]; // Remove sufixos como "-02" e converte vírgula em ponto para float let lapTexto = celula.split('-')[0].trim(); let tempoNum = parseFloat(lapTexto.replace(',', '.')); if (!isNaN(tempoNum) && tempoNum > 0) { pilotosMap[pName].laps[numVolta - 1] = tempoNum; } } }); } }); let resultadoFinal = []; let coresBadge = ["#ffb703", "#e76f51", "#2ec4b6", "#3a86ff", "#e63946"]; let contador = 0; pilotoNomes.forEach(pName => { let pObj = pilotosMap[pName]; // Remove buracos mantendo a sequência exata das voltas pObj.laps = pObj.laps.filter(val => val !== undefined && !isNaN(val)); if (pObj.laps.length > 0) { let melhor = Math.min(...pObj.laps); let soma = pObj.laps.reduce((a, b) => a + b, 0); let media = soma / pObj.laps.length; let variancia = pObj.laps.reduce((sum, t) => sum + Math.pow(t - media, 2), 0) / pObj.laps.length; let desvio = Math.sqrt(variancia); pObj.melhorVoltaVal = parseFloat(melhor.toFixed(3)); pObj.melhorVoltaTxt = melhor.toFixed(3).replace('.', ',') + 's'; pObj.mediaVal = parseFloat(media.toFixed(3)); pObj.mediaTxt = media.toFixed(3).replace('.', ',') + 's'; pObj.desvioVal = parseFloat(desvio.toFixed(3)); pObj.desvio = `±${desvio.toFixed(3).replace('.', ',')}s`; pObj.voltasTotais = pObj.laps.length; } pObj.badgeColor = coresBadge[contador % coresBadge.length]; contador++; resultadoFinal.push(pObj); }); return resultadoFinal; }
